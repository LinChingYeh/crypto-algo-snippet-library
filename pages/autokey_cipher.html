自動金鑰密碼 (Autokey Cipher)

原理簡介

自動金鑰密碼是一種多表替換密碼，由法國密碼學家布萊斯·德·維吉尼亞（Blaise de Vigenère）發明，目的是為了解決傳統維吉尼亞密碼（Vigenère cipher）密鑰重複出現導致容易被頻率分析破解的弱點。它的特點是密鑰由兩部分組成：第一部分是用戶設定的「初始密鑰（Keyword）」，第二部分則是直接將「明文」本身接在初始密鑰之後作為後續的密鑰。因此，密鑰的長度會自動與明文等長，且不會出現規律的週期性重複，大幅提升了加密的安全性。

複雜度與效能分析

  • 時間複雜度：O(N) (N 為輸入字串的長度。演算法需要走訪整個文字字串，對每個字母進行模運算，處理時間隨字串長度線性增加)
  • 空間複雜度：O(N) (因為需要動態將明文加入 `keyStream` 向量中作為後續的密鑰，所以需要配置與字串長度成正比的額外記憶體空間)

核心程式碼

#include <iostream>
#include <string>
#include <vector>
#include <limits>
using namespace std;

// 自動密鑰密碼函式
string autokeyCipher(string text, string keyword, bool decrypt) {

    // 儲存密鑰字母對應的數值
    vector<int> keyStream;

    // 將初始密鑰轉換成 0~25
    for (int i = 0; i < keyword.length(); i++) {

        if (keyword[i] >= 'A' && keyword[i] <= 'Z') {
            keyStream.push_back(keyword[i] - 'A');
        }
        else if (keyword[i] >= 'a' && keyword[i] <= 'z') {
            keyStream.push_back(keyword[i] - 'a');
        }
    }

    // 記錄目前使用到第幾個密鑰
    int keyIndex = 0;

    // 逐一處理文字中的字元
    for (int i = 0; i < text.length(); i++) {

        // 處理大寫英文字母
        if (text[i] >= 'A' && text[i] <= 'Z') {

            int p = text[i] - 'A';
            int k = keyStream[keyIndex];
            int result;

            if (decrypt) {
                // 解密公式：P = (C - K) mod 26
                result = (p - k + 26) % 26;
            }
            else {
                // 加密公式：C = (P + K) mod 26
                result = (p + k) % 26;
            }

            // 將運算結果轉回英文字母
            text[i] = 'A' + result;

            // 將明文加入後續密鑰
            keyStream.push_back(decrypt ? result : p);

            keyIndex++;
        }

        // 處理小寫英文字母
        else if (text[i] >= 'a' && text[i] <= 'z') {

            int p = text[i] - 'a';
            int k = keyStream[keyIndex];
            int result;

            if (decrypt) {
                // 解密公式：P = (C - K) mod 26
                result = (p - k + 26) % 26;
            }
            else {
                // 加密公式：C = (P + K) mod 26
                result = (p + k) % 26;
            }

            // 將運算結果轉回英文字母
            text[i] = 'a' + result;

            // 將明文加入後續密鑰
            keyStream.push_back(decrypt ? result : p);

            keyIndex++;
        }
    }

    return text;
}

int main() {
    int choice;
    string keyword;
    string text;

    cout << "===== Autokey Cipher =====" << endl;
    cout << "1. Encryption" << endl;
    cout << "2. Decryption" << endl;

    cout << "Choose an option: ";
    cin >> choice;

    // 檢查選項
    if (choice != 1 && choice != 2) {
        cout << "Invalid option!" << endl;
        return 0;
    }

    // 清除輸入緩衝區
    cin.ignore(numeric_limits<streamsize>::max(), '\n');

    cout << "Enter Keyword: ";
    getline(cin, keyword);

    // 檢查密鑰是否至少包含一個英文字母
    bool validKey = false;

    for (int i = 0; i < keyword.length(); i++) {
        if ((keyword[i] >= 'A' && keyword[i] <= 'Z') ||
            (keyword[i] >= 'a' && keyword[i] <= 'z')) {
            validKey = true;
            break;
        }
    }

    if (!validKey) {
        cout << "Error: Keyword must contain at least one letter!" << endl;
        return 0;
    }

    cout << "Enter text: ";
    getline(cin, text);

    // 執行加密或解密
    if (choice == 1) {
        cout << "Encrypted result: "
             << autokeyCipher(text, keyword, false) << endl;
    }
    else {
        cout << "Decrypted result: "
             << autokeyCipher(text, keyword, true) << endl;
    }

    return 0;
}
